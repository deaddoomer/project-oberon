  PROCEDURE SDLBlitSurface(src, sx, sy, sw, sh, dst, dx, dy, dw, dh: INTEGER): INTEGER;
    VAR sr, dr: RECORD xy, wh: INTEGER END;
  BEGIN
    sr.xy := sy * 65536 + sx MOD 65536;
    sr.wh := sh * 65536 + sw MOD 65536;
    dr.xy := dy * 65536 + dx MOD 65536;
    dr.wh := dh * 65536 + dw MOD 65536;
    RETURN sdlBlitSurface(SYSTEM.ADR(dr), dst, SYSTEM.ADR(sr), src)
  END SDLBlitSurface;

  PROCEDURE UnixToClock(t: INTEGER): INTEGER;
    VAR z, era, y, doe, yoe, doy, mp, second, minute, hour, day, month, year: INTEGER;
  BEGIN
    z := (t DIV 86400) + 719468;
    IF z >= 0 THEN era := z DIV 146097 ELSE era := (z - 146096) DIV 146097 END;
    doe := z - era*146097;
    yoe := (doe - doe DIV 1460 + doe DIV 36524 - doe DIV 146096) DIV 365;
    doy := doe - (365*yoe + yoe DIV 4 - yoe DIV 100);
    mp := (5*doy + 2) DIV 153;
    y := yoe + era*400;
    day := doy - (153*mp + 2) DIV 5 + 1;
    IF mp < 10 THEN month := mp + 3 ELSE month := mp - 9 END;
    IF month <= 2 THEN year := y + 1 ELSE year := y END;
    year := year MOD 100;
    hour := t DIV 3600 MOD 24;
    minute := t DIV 60 MOD 60;
    second := t MOD 60;
    RETURN ((((year*10H + month)*20H + day)*20H + hour)*40H + minute)*40H + second
  END UnixToClock;

  PROCEDURE FileInfo*(name: ARRAY OF CHAR; VAR length, date: INTEGER);
    VAR res: INTEGER; statbuf: ARRAY 24 OF INTEGER; 
  BEGIN length := -1; date := 0;
    res := syscall3(195, SYSTEM.ADR(name[0]), SYSTEM.ADR(statbuf)); (*stat64*)
    IF res = 0 THEN
      IF (statbuf[12] = 0) & (statbuf[11] >= 0) THEN length := statbuf[11] END; (*st_size*)
      date := UnixToClock(statbuf[20]) (*st_ctime*)
    END
  END FileInfo;

  PROCEDURE Write*(fd, buf, size: INTEGER): INTEGER;
    VAR len: INTEGER;
  BEGIN
    REPEAT len := syscall4(4, fd, buf, size) UNTIL len # -4;
    RETURN len
  END Write;

  PROCEDURE WriteStringAdr*(fd, p: INTEGER);
    VAR len: INTEGER; ch: CHAR;
  BEGIN
    IF p # 0 THEN
      len := 0; SYSTEM.GET(p, ch); WHILE ch # 0X DO INC(len); SYSTEM.GET(p+len, ch) END;
      len := Write(fd, p, len)
    END
  END WriteStringAdr;

  PROCEDURE WriteString*(fd: INTEGER; str: ARRAY OF CHAR);
  BEGIN WriteStringAdr(fd, SYSTEM.ADR(str))
  END WriteString;

  PROCEDURE WriteLn*(fd: INTEGER);
    VAR s: ARRAY 2 OF CHAR;
  BEGIN s[0] := 0AX; s[1] := 0X; WriteString(fd, s)
  END WriteLn;

  (* --------- signals --------- *)

  PROCEDURE Lock;
    VAR m: INTEGER;
  BEGIN
    REPEAT m := syscall1(68) UNTIL m # -4; (*siggetmask()*)
    REPEAT UNTIL syscall2(69, ORD(SYSTEM.VAL(SET, m) - {2})) # -4; (*sigsetmask(m-2)*)
  END Lock;

  PROCEDURE Unlock;
    VAR m: INTEGER;
  BEGIN
    REPEAT m := syscall1(68) UNTIL m # -4; (*siggetmask()*)
    REPEAT UNTIL syscall2(69, ORD(SYSTEM.VAL(SET, m) + {2})) # -4; (*sigsetmask(m+2)*)
  END Unlock;

  PROCEDURE Abort*;
    VAR res, pid: INTEGER;
  BEGIN
    Lock;
    quit := TRUE;
    IF thread # 0 THEN SDLWaitThread(thread, res) END;
    REPEAT pid := syscall1(20) UNTIL pid # -4; (*getpid()*)
    REPEAT UNTIL syscall3(37, pid, 6) # -4; (*kill(getpid(), SIGABRT)*)
    REPEAT UNTIL syscall2(1, 1) # -4; (*exit(1)*)
    REPEAT UNTIL FALSE
  END Abort;

  PROCEDURE Signal; (*void Signal(int sig, struct sigcontext ctx);*)
    VAR AbortCommand: PROCEDURE;
  BEGIN
    SYSTEM.GET(Modules.MTOrg - 32, AbortCommand);
    IF AbortCommand # NIL THEN
      SYSTEM.PUT(SYSTEM.REG(5) + 12 + 56, AbortCommand); (*ctx.eip := AbortCommand*)
    ELSE Abort
    END;
  END Signal;

  PROCEDURE InstallSignals;
    VAR sa: RECORD
      handler: INTEGER;
      mask, flags: SET;
      restorer: INTEGER;
    END;
  BEGIN
    sa.handler := SYSTEM.ADR(Signal);
    sa.mask := {};
    sa.flags := {28, 30}; (*SA_RESTART|SA_NODEFER*)
    sa.restorer := 0;
    REPEAT UNTIL syscall4(67, 2, SYSTEM.ADR(sa), 0) # -4; (*sigaction(SIGINT, &sa, NULL)*)
  END InstallSignals;

  (* --------- commands --------- *)

  PROCEDURE Sleep*(ms: INTEGER);
  BEGIN SDLDelay(ms)
  END Sleep;

  PROCEDURE Idle*;
  BEGIN SDLDelay(1)
  END Idle;

  PROCEDURE Quit*(code: INTEGER);
    VAR res: INTEGER; ok: BOOLEAN;
  BEGIN
    Lock;
    quit := TRUE;
    IF thread # 0 THEN SDLWaitThread(thread, res) END;
    REPEAT UNTIL syscall1(36) # -4; (*sync*)
    IF SDL # 0 THEN SDLQuit; Libraries.Free(SDL); SDL := 0 END;
    REPEAT UNTIL syscall2(1, code) # -4; (*exit(code)*)
    REPEAT UNTIL FALSE
  END Quit;

  PROCEDURE Exit*;
  BEGIN Quit(0)
  END Exit;

  PROCEDURE Sync*;
  BEGIN REPEAT UNTIL syscall1(36) # -4; (*sync*)
  END Sync;

  (* --------- display --------- *)

  PROCEDURE GetBuffer*(VAR base, width, height, pitch, depth: INTEGER; askDepth: INTEGER);
    VAR arg: ARRAY 6 OF INTEGER;
  BEGIN base := -1;
    IF (askDepth > 0) & (askDepth <= 32) & ((askDepth = 32) OR (askDepth IN {1, 8, 16, 24})) THEN
      base := 0; width := 1024; height := 768; depth := askDepth; pitch := ((depth + 7) DIV 8) * width;

      (* base = mmap(0, pitch * height, PROT_READ | PROT_WRITE, MAP_PRIVATE | MAP_ANONYMOUS, -1, 0); *)
      arg[0] := 0; arg[1] := pitch * height; arg[2] := 3; arg[3] := 2H + 20H; arg[4] := -1; arg[5] := 0;
      REPEAT base := syscall2(90, SYSTEM.ADR(arg)) UNTIL base # -4;
      IF (base >= -4096) & (base < 0) THEN base := -1 END;
    END;

    IF base = -1 THEN
      base := 0; width := 0; height := 0; pitch := 0; depth := 0;
    END;
  END GetBuffer;

  PROCEDURE SetBuffer*(base, width, height, scanline, bpp: INTEGER);
    VAR ev: SDLEvent;
  BEGIN
    ev.type := SDLUserEvent;
    ev.pad0 := bpp;
    ev.any[0] := base;
    ev.any[1] := width;
    ev.any[2] := height;
    ev.any[3] := scanline;
    ev.any[4] := 0;
    Lock; REPEAT UNTIL SDLPushEvent(SYSTEM.ADR(ev)); Unlock;
  END SetBuffer;

  PROCEDURE Repaint*(x, y, w, h: INTEGER);
  BEGIN repaint := TRUE;
  END Repaint;

  (* --------- input --------- *)

  PROCEDURE Available*(): INTEGER;
    VAR Recd: BOOLEAN;
  BEGIN
    IF quit THEN Exit END;
    Lock;
    REPEAT UNTIL SDLLockMutex(mutex);
    Recd := KI # KJ;
    REPEAT UNTIL SDLUnlockMutex(mutex);
    Unlock;
    RETURN ORD(Recd)
  END Available;

  PROCEDURE Read*(VAR ch: CHAR);
  BEGIN
    IF quit THEN Exit END;
    Lock; REPEAT UNTIL SDLLockMutex(mutex);
    WHILE KI = KJ DO (*wait input*)
      REPEAT UNTIL SDLUnlockMutex(mutex); Unlock; (*unlock signal in case of sigint*)
      Lock; SDLDelay(10); Unlock;
      Lock; REPEAT UNTIL SDLLockMutex(mutex);
    END;
    ch := KBuf[KI]; KI := (KI + 1) MOD kbdsize;
    REPEAT UNTIL SDLUnlockMutex(mutex); Unlock;
  END Read;

  PROCEDURE Mouse*(VAR keys: SET; VAR x, y: INTEGER);
    VAR bits: SET; n: INTEGER;
  BEGIN
    IF quit THEN Exit END;
    Lock;
    REPEAT UNTIL SDLLockMutex(mutex);
    keys := MK; x := MX; y := MY;
    REPEAT UNTIL SDLUnlockMutex(mutex);
    Unlock;
  END Mouse;

  PROCEDURE Loop;
    VAR
      pid: INTEGER;
      ev: SDLEvent; window, fb, temp, Width, Height: INTEGER;
      kev: SDLKeyboardEvent; kbdCode: INTEGER; Shift, Ctrl: BOOLEAN; ch: CHAR;
      bits, keys: SET; x, y: INTEGER; scanMouse: BOOLEAN;
      active: BOOLEAN;
      (*SHARED GLOBAL VARIABLES: thread, mutex, quit, repaint, MK, MX, MY, KI, KJ, KBuf*)
  BEGIN
    REPEAT pid := syscall1(20) UNTIL pid # -4; (*pid := getpid()*)
    REPEAT UNTIL syscall4(172, 15, SYSTEM.ADR("SDL Oberon loop"), 0) # -4; (*prctl(RP_SET_NAME, "SDL Oberon loop", 0, ?, ?)*)
    window := 0; fb := 0; repaint := FALSE; active := FALSE;
    SDLEnableKeyRepeat(500, 30);
    SDLShowCursor(0);

    REPEAT
      IF SDLPollEvent(SYSTEM.ADR(ev)) THEN
        IF ev.type = 1 THEN (*SDL_ACTIVEEVENT*)
          IF ev.pad1 = 4 THEN (*SDL_APPACTIVE*)
            active := ev.pad0 # 0;
            IF active THEN
              repaint := TRUE;
            END;
          END;
        ELSIF ev.type = 2 THEN (*SDL_KEYDOWN*)
          kev := SYSTEM.VAL(SDLKeyboardEvent, ev);
          kbdCode := kev.keysym.sym;
          Shift := (kev.keysym.mod * {0, 1}) # {}; (*shift*)
          Ctrl := 6 IN kev.keysym.mod; (*ctrl*)
          IF kbdCode < 128 THEN
            IF Shift OR Ctrl THEN INC(kbdCode, 80H) END; (*ctrl implies shift*)
            SYSTEM.GET(KTabAdr + kbdCode, ch);
          ELSIF kbdCode = 282 THEN (*f1*)
            IF Shift OR Ctrl THEN ch := 0X ELSE ch := 1AX END;
          ELSIF kbdCode = 266 THEN (*keypad del*)
            ch := 7FX (*del*)
          ELSE ch := 0X
          END;
          IF (7 IN kev.keysym.mod) & ((kbdCode = 318) OR (kbdCode = 19)) THEN (*lctrl+break/pause*)
            REPEAT UNTIL syscall3(37, pid, 2) # -4; (*kill(pid, SIGINT)*)
          ELSE (*push into keyboard buffer*)
            IF Ctrl THEN ch := CHR(ORD(ch) MOD 20H) END;
            REPEAT UNTIL SDLLockMutex(mutex);
            temp := (KJ + 1) MOD kbdsize;
            IF temp = KI THEN
              WriteString(stderr, "Warning: keyboard buffer overrun"); WriteLn(stderr);
              KI := (KI + 1) MOD kbdsize; (*drop last char*)
            END;
            KBuf[temp] := ch; KI := temp;
            REPEAT UNTIL SDLUnlockMutex(mutex);
          END;
        ELSIF (ev.type IN {4, 5, 6}) THEN (*SDL_MOUSEMOTION, SDL_MOUSEBUTTONDOWN, SDL_MOUSEBUTTONUP*)
          scanMouse := TRUE;
        ELSIF ev.type = 12 THEN (*SDL_QUIT*)
          quit := TRUE;
        ELSIF ev.type = 17 THEN (*SDL_VIDEOEXPOSE*)
          repaint := TRUE;
        ELSIF ev.type = SDLUserEvent+0 THEN (*SetBuffer*)
          temp := SDLSetVideoMode(ev.any[1], ev.any[2], ev.pad0, {anyformat});
          IF temp # 0 THEN
            SDLWMSetCaption("Oberon", "Oberon");
            window := temp;
          ELSE WriteString(stderr, "Error: SDL_SetVideoMode: "); WriteStringAdr(stderr, SDLGetError()); WriteLn(stderr);
          END;
          temp := SDLCreateRGBSurfaceFrom(ev.any[0], ev.any[1], ev.any[2], ev.pad0, ev.any[3], 0, 0, 0, 0);
          IF temp # 0 THEN
            IF fb # 0 THEN SDLFreeSurface(fb) END;
            fb := temp;
          ELSE WriteString(stderr, "Error: SDL_CreateRGBSurface: "); WriteStringAdr(stderr, SDLGetError()); WriteLn(stderr);
          END;
          active := (window # 0) & (fb # 0); repaint := TRUE;
          IF fb # 0 THEN
            SYSTEM.GET(fb + 8, Width);
            SYSTEM.GET(fb + 12, Height);
          END;
        END;
      END;

      IF scanMouse THEN
        keys := {}; bits := SDLGetMouseState(x, y);
        IF (*0*) 0 IN bits THEN INCL(keys, 2) END; (*left*)
        IF (*1*) 2 IN bits THEN INCL(keys, 0) END; (*middle*)
        IF (*2*) 1 IN bits THEN INCL(keys, 1) END; (*right*)
        REPEAT UNTIL SDLLockMutex(mutex);
        MK := keys; MX := x; MY := Height - y;
        REPEAT UNTIL SDLUnlockMutex(mutex);
        scanMouse := FALSE;
      END;

      (*redraw screen*)
      IF active THEN
        IF repaint & (window # 0) & (fb # 0) THEN
          repaint := FALSE; (*possible race condition here, but it hint anyway*)
          REPEAT UNTIL SDLBlitSurface(fb, 0, 0, Width, Height, window, 0, 0, Width, Height) = 0;
          SDLFlip(window);
        END;
        SDLDelay(1);
      ELSE SDLDelay(10);
      END
    UNTIL quit;
  END Loop;

  BEGIN

      00 00 00 00 00 00 00 00  00 00 00 1B 00 00 00 00

      00 00 00 00 00 00 00 00  08 09 00 00 00 0D 00 00

        SYSTEM.PUT(SYSTEM.ADR(sdlFreeSurface), Libraries.ThisProc(SDL, "SDL_FreeSurface"));
        SYSTEM.PUT(SYSTEM.ADR(sdlBlitSurface), Libraries.ThisProc(SDL, "SDL_UpperBlit")); (*SDL_BlitSurface*)
        SYSTEM.PUT(SYSTEM.ADR(sdlLockSurface), Libraries.ThisProc(SDL, "SDL_LockSurface"));
        SYSTEM.PUT(SYSTEM.ADR(sdlUnlockSurface), Libraries.ThisProc(SDL, "SDL_UnlockSurface"));
        SYSTEM.PUT(SYSTEM.ADR(SDLGetError), Libraries.ThisProc(SDL, "SDL_GetError"));
        SYSTEM.PUT(SYSTEM.ADR(sdlWMSetCaption), Libraries.ThisProc(SDL, "SDL_WM_SetCaption"));
        SYSTEM.PUT(SYSTEM.ADR(SDLPumpEvents), Libraries.ThisProc(SDL, "SDL_PumpEvents"));
        SYSTEM.PUT(SYSTEM.ADR(sdlGetMouseState), Libraries.ThisProc(SDL, "SDL_GetMouseState"));
        SYSTEM.PUT(SYSTEM.ADR(sdlShowCursor), Libraries.ThisProc(SDL, "SDL_ShowCursor"));
        SYSTEM.PUT(SYSTEM.ADR(sdlPeepEvents), Libraries.ThisProc(SDL, "SDL_PeepEvents"));
        SYSTEM.PUT(SYSTEM.ADR(sdlWaitEvent), Libraries.ThisProc(SDL, "SDL_WaitEvent"));
        SYSTEM.PUT(SYSTEM.ADR(sdlEnableKeyRepeat), Libraries.ThisProc(SDL, "SDL_EnableKeyRepeat"));
        SYSTEM.PUT(SYSTEM.ADR(sdlCreateThread), Libraries.ThisProc(SDL, "SDL_CreateThread"));
        SYSTEM.PUT(SYSTEM.ADR(sdlWaitThread), Libraries.ThisProc(SDL, "SDL_WaitThread"));
        SYSTEM.PUT(SYSTEM.ADR(SDLCreateMutex), Libraries.ThisProc(SDL, "SDL_CreateMutex"));
        SYSTEM.PUT(SYSTEM.ADR(sdlLockMutex), Libraries.ThisProc(SDL, "SDL_mutexP")); (*SDL_LockMutex*)
        SYSTEM.PUT(SYSTEM.ADR(sdlUnlockMutex), Libraries.ThisProc(SDL, "SDL_mutexV")); (*SDL_LockMutex*)
        SYSTEM.PUT(SYSTEM.ADR(sdlDestroyMutex), Libraries.ThisProc(SDL, "SDL_DestroyMutex"));
        SYSTEM.PUT(SYSTEM.ADR(sdlPushEvent), Libraries.ThisProc(SDL, "SDL_PushEvent"));
        SYSTEM.PUT(SYSTEM.ADR(sdlDelay), Libraries.ThisProc(SDL, "SDL_Delay"));
        SYSTEM.PUT(SYSTEM.ADR(sdlFlip), Libraries.ThisProc(SDL, "SDL_Flip"));
        IF SDLInit({video, noparachute}) = -1 THEN
          WriteString(stderr, "Error: SDL_Init: "); WriteStringAdr(stderr, SDLGetError()); WriteLn(stderr);
          Abort
        END;
        mutex := SDLCreateMutex();
        IF mutex = 0 THEN
          WriteString(stderr, "Error: SDL_CreateMutex: "); WriteStringAdr(stderr, SDLGetError()); WriteLn(stderr);
          Abort
        END;
        thread := SDLCreateThread(Loop, 0);
        IF thread = 0 THEN
          WriteString(stderr, "Error: SDL_CreateThread: "); WriteStringAdr(stderr, SDLGetError()); WriteLn(stderr);
          Abort
        END;
      ELSE
        WriteString(stderr, "Error: failed to load "); WriteString(stderr, library); WriteLn(stderr);
        Abort
      END
    END;
    Unlock
  END Init;
