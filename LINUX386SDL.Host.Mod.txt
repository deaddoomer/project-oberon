MODULE Host; (*linux/386 + SDL-1.2*)
  IMPORT SYSTEM, Libraries;

  CONST
    library = "libSDL.so";
    stderr = 2;

    (*SDLInit*)
    timer = 1; video = 5; noparachute = 20;

    (*SDLSetVideoMode*)
    resizable = 4; anyformat = 28;

  TYPE
    SDLEvent = RECORD
      type, pad0, pad1, pad2: BYTE;
      any: ARRAY 20 OF BYTE;
    END;

  VAR
    maxw*, maxh*: INTEGER;
    surface, dbuf: INTEGER;
    active: BOOLEAN;
    Base, Width, Height, Pitch: INTEGER;

    syscall1: PROCEDURE (eax: INTEGER): INTEGER;
    syscall2: PROCEDURE (eax, ebx: INTEGER): INTEGER;
    syscall3: PROCEDURE (eax, ebx, ecx: INTEGER): INTEGER;
    syscall4: PROCEDURE (eax, ebx, ecx, edx: INTEGER): INTEGER;

    SDL*: INTEGER;
    SDLQuit: PROCEDURE (*ccall*) ();
    sdlInit: PROCEDURE (*ccall*) (flags: SET): INTEGER;
    sdlSetVideoMode: PROCEDURE (*ccall*) (flags: SET; bpp, height, width: INTEGER): INTEGER;
    sdlUpdateRect: PROCEDURE (*ccall*) (h, w, y, x, screen: INTEGER);
    sdlPollEvent: PROCEDURE (*ccall*) (ev: INTEGER): INTEGER;
    sdlCreateRGBSurfaceFrom: PROCEDURE (*ccall*) (Amask, Bmask, Gmask, Rmask, pitch, depth, height, width, pixels: INTEGER): INTEGER;
    sdlFreeSurface: PROCEDURE (*ccall*) (surface: INTEGER);
    sdlBlitSurface: PROCEDURE (*ccall*) (dstrect, dst, srcrect, src: INTEGER): INTEGER;
    sdlLockSurface: PROCEDURE (*ccall*) (surface: INTEGER): INTEGER;
    sdlUnlockSurface: PROCEDURE (*ccall*) (surface: INTEGER);
    sdlWMSetCaption: PROCEDURE (*ccall*) (icon, title: INTEGER);
    SDLGetError: PROCEDURE (*ccall*) (): INTEGER;

  PROCEDURE SDLInit (flags: SET): INTEGER;
  BEGIN RETURN sdlInit(flags)
  END SDLInit;

  PROCEDURE SDLLockSurface(surface: INTEGER): BOOLEAN;
  BEGIN RETURN sdlLockSurface(surface) = 0
  END SDLLockSurface;

  PROCEDURE SDLUnlockSurface(surface: INTEGER);
  BEGIN sdlUnlockSurface(surface)
  END SDLUnlockSurface;

  PROCEDURE SDLWMSetCaption(title, icon: ARRAY OF CHAR);
  BEGIN sdlWMSetCaption(SYSTEM.ADR(icon), SYSTEM.ADR(title))
  END SDLWMSetCaption;

  PROCEDURE SDLSetVideoMode(width, height, bpp: INTEGER; flags: SET): INTEGER;
  BEGIN RETURN sdlSetVideoMode(flags, bpp, height, width)
  END SDLSetVideoMode;

  PROCEDURE SDLUpdateRect(surface, x, y, w, h: INTEGER);
  BEGIN sdlUpdateRect(h, w, y, x, surface)
  END SDLUpdateRect;

  PROCEDURE SDLCreateRGBSurfaceFrom (pixels, width, height, depth, pitch, Rmask, Gmask, Bmask, Amask: INTEGER): INTEGER;
  BEGIN RETURN sdlCreateRGBSurfaceFrom(Amask, Bmask, Gmask, Rmask, pitch, depth, height, width, pixels)
  END SDLCreateRGBSurfaceFrom;

  PROCEDURE SDLFreeSurface(surface: INTEGER);
  BEGIN sdlFreeSurface(surface)
  END SDLFreeSurface;

  PROCEDURE SDLBlitSurface(src, sx, sy, sw, sh, dst, dx, dy, dw, dh: INTEGER): INTEGER;
    VAR sr, dr: RECORD xy, wh: INTEGER END;
  BEGIN
    sr.xy := sy * 65536 + sx MOD 65536;
    sr.wh := sh * 65536 + sw MOD 65536;
    dr.xy := dy * 65536 + dx MOD 65536;
    dr.wh := dh * 65536 + dw MOD 65536;
    RETURN sdlBlitSurface(SYSTEM.ADR(dr), dst, SYSTEM.ADR(sr), src)
  END SDLBlitSurface;

  PROCEDURE SDLPollEvent (VAR ev: SDLEvent): BOOLEAN;
  BEGIN RETURN sdlPollEvent(SYSTEM.ADR(ev.type)) # 0
  END SDLPollEvent;

  PROCEDURE Quit*(code: INTEGER);
    VAR res: INTEGER; ok: BOOLEAN;
  BEGIN
    res := syscall1(36); (*sync*)
    IF SDL # 0 THEN SDLQuit; Libraries.Free(SDL); SDL := 0 END;
    res := syscall2(1, code) (*exit(code)*)
  END Quit;

  PROCEDURE Exit*;
  BEGIN Quit(0)
  END Exit;

  PROCEDURE Sync*;
    VAR res: INTEGER;
  BEGIN res := syscall1(36) (*sync*)
  END Sync;

  PROCEDURE Abort*;
    VAR res: INTEGER;
  BEGIN
    res := syscall3(37, syscall1(20), 6); (*kill(getpid(), SIGABRT)*)
    res := syscall2(1, 1); (*exit*)
    REPEAT UNTIL FALSE
  END Abort;

  PROCEDURE UnixToClock(t: INTEGER): INTEGER;
    VAR z, era, y, doe, yoe, doy, mp, second, minute, hour, day, month, year: INTEGER;
  BEGIN
    z := (t DIV 86400) + 719468;
    IF z >= 0 THEN era := z DIV 146097 ELSE era := (z - 146096) DIV 146097 END;
    doe := z - era*146097;
    yoe := (doe - doe DIV 1460 + doe DIV 36524 - doe DIV 146096) DIV 365;
    doy := doe - (365*yoe + yoe DIV 4 - yoe DIV 100);
    mp := (5*doy + 2) DIV 153;
    y := yoe + era*400;
    day := doy - (153*mp + 2) DIV 5 + 1;
    IF mp < 10 THEN month := mp + 3 ELSE month := mp - 9 END;
    IF month <= 2 THEN year := y + 1 ELSE year := y END;
    year := year MOD 100;
    hour := t DIV 3600 MOD 24;
    minute := t DIV 60 MOD 60;
    second := t MOD 60;
    RETURN ((((year*10H + month)*20H + day)*20H + hour)*40H + minute)*40H + second
  END UnixToClock;

  PROCEDURE FileInfo*(name: ARRAY OF CHAR; VAR length, date: INTEGER);
    VAR res: INTEGER; statbuf: ARRAY 24 OF INTEGER; 
  BEGIN length := -1; date := 0;
    res := syscall3(195, SYSTEM.ADR(name[0]), SYSTEM.ADR(statbuf)); (*stat64*)
    IF res = 0 THEN
      IF (statbuf[12] = 0) & (statbuf[11] >= 0) THEN length := statbuf[11] END; (*st_size*)
      date := UnixToClock(statbuf[20]) (*st_ctime*)
    END
  END FileInfo;

  PROCEDURE Write*(fd, buf, size: INTEGER): INTEGER;
    VAR len: INTEGER;
  BEGIN
    REPEAT len := syscall4(4, fd, buf, size) UNTIL len # -4;
    RETURN len
  END Write;

  PROCEDURE WriteStringAdr*(fd, p: INTEGER);
    VAR len: INTEGER; ch: CHAR;
  BEGIN
    IF p # 0 THEN
      len := 0; SYSTEM.GET(p, ch); WHILE ch # 0X DO INC(len); SYSTEM.GET(p+len, ch) END;
      len := Write(fd, p, len)
    END
  END WriteStringAdr;

  PROCEDURE WriteString*(fd: INTEGER; str: ARRAY OF CHAR);
  BEGIN WriteStringAdr(fd, SYSTEM.ADR(str))
  END WriteString;

  PROCEDURE WriteLn*(fd: INTEGER);
    VAR s: ARRAY 2 OF CHAR;
  BEGIN s[0] := 0AX; s[1] := 0X; WriteString(fd, s)
  END WriteLn;

  PROCEDURE SetBuffer*(base, width, height, scanline, bpp: INTEGER);
  BEGIN
    active := FALSE;
    (*surface := SDLSetVideoMode(width, height, bpp, {anyformat});*)
    surface := SDLSetVideoMode(width, height, 0, {anyformat});
    Base := base; Width := width; Height := height; Pitch := scanline;
    IF surface = 0 THEN
      WriteString(stderr, "Error: SDL_SetVideoMode: "); WriteStringAdr(stderr, SDLGetError()); WriteLn(stderr);
      Abort
    END;
    IF dbuf # 0 THEN SDLFreeSurface(dbuf) END;
    dbuf := SDLCreateRGBSurfaceFrom(Base, Width, Height, bpp, Pitch, 0, 0, 0, 0);
    IF dbuf = 0 THEN
      WriteString(stderr, "Error: SDL_CreateRGBSurfaceFrom: "); WriteStringAdr(stderr, SDLGetError()); WriteLn(stderr);
      Abort
    END;
    SDLWMSetCaption("Oberon", "Oberon");
    active := TRUE
  END SetBuffer;

  PROCEDURE Repaint*(x, y, w, h: INTEGER);
    VAR Y: INTEGER;
  BEGIN
    IF active & (surface # 0) & (w > 0) & (h > 0) & (dbuf # 0) THEN
      Y := Height - y - h;
      REPEAT UNTIL SDLBlitSurface(dbuf, x, Y, w, h, surface, x, Y, Width, Height) = 0;
      SDLUpdateRect(surface, x, Y, w, h)
    END
  END Repaint;

  PROCEDURE PollEvent*;
    VAR ev: SDLEvent;
  BEGIN
    IF SDLPollEvent(ev) THEN
      IF ev.type = 1 THEN (*SDL_ACTIVEEVENT*)
        IF ev.pad1 = 4 THEN (*SDL_APPACTIVE*)
          IF ev.pad0 # 0 THEN active := TRUE; Repaint(0, 0, Width, Height) END
        END
      ELSIF ev.type = 12 THEN (*SDL_QUIT*)
        Exit
      ELSIF ev.type = 17 THEN (*SDL_VIDEOEXPOSE*)
        Repaint(0, 0, Width, Height)
      END
    END
  END PollEvent;

  PROCEDURE Init;
    VAR res: INTEGER; ok, done: BOOLEAN;
  BEGIN
    maxw := 1024; maxh := 768; active := FALSE;
    SYSTEM.PUT(SYSTEM.ADR(syscall1), SYSTEM.ADR($8B442404 CD80 C20400 6690 90$));
    SYSTEM.PUT(SYSTEM.ADR(syscall2), SYSTEM.ADR($8B442408 8B5C2404 CD80 C20800 6690 90$));
    SYSTEM.PUT(SYSTEM.ADR(syscall3), SYSTEM.ADR($8B44240C 8B5C2408 8B4C2404 CD80 C20C00 6690 90$));
    SYSTEM.PUT(SYSTEM.ADR(syscall4), SYSTEM.ADR($8B442410 8B5C240C 8B4C2408 8B542404 CD80 C21000 6690 90$));
    IF SDL = 0 THEN
      Libraries.Load(library, SDL);
      IF SDL # 0 THEN
        SYSTEM.PUT(SYSTEM.ADR(sdlInit), Libraries.ThisProc(SDL, "SDL_Init"));
        SYSTEM.PUT(SYSTEM.ADR(SDLQuit), Libraries.ThisProc(SDL, "SDL_Quit"));
        SYSTEM.PUT(SYSTEM.ADR(sdlSetVideoMode), Libraries.ThisProc(SDL, "SDL_SetVideoMode"));
        SYSTEM.PUT(SYSTEM.ADR(sdlUpdateRect), Libraries.ThisProc(SDL, "SDL_UpdateRect"));
        SYSTEM.PUT(SYSTEM.ADR(sdlPollEvent), Libraries.ThisProc(SDL, "SDL_PollEvent"));
        SYSTEM.PUT(SYSTEM.ADR(sdlCreateRGBSurfaceFrom), Libraries.ThisProc(SDL, "SDL_CreateRGBSurfaceFrom"));
        SYSTEM.PUT(SYSTEM.ADR(sdlFreeSurface), Libraries.ThisProc(SDL, "SDL_FreeSurface"));
        SYSTEM.PUT(SYSTEM.ADR(sdlBlitSurface), Libraries.ThisProc(SDL, "SDL_UpperBlit")); (*SDL_BlitSurface*)
        SYSTEM.PUT(SYSTEM.ADR(sdlLockSurface), Libraries.ThisProc(SDL, "SDL_LockSurface"));
        SYSTEM.PUT(SYSTEM.ADR(sdlUnlockSurface), Libraries.ThisProc(SDL, "SDL_UnlockSurface"));
        SYSTEM.PUT(SYSTEM.ADR(SDLGetError), Libraries.ThisProc(SDL, "SDL_GetError"));
        SYSTEM.PUT(SYSTEM.ADR(sdlWMSetCaption), Libraries.ThisProc(SDL, "SDL_WM_SetCaption"));
        IF SDLInit({video, noparachute}) = -1 THEN
          WriteString(stderr, "Error: SDL_Init: "); WriteStringAdr(stderr, SDLGetError()); WriteLn(stderr);
          Abort
        END
      ELSE
        WriteString(stderr, "Error: failed to load "); WriteString(stderr, library); WriteLn(stderr);
        Abort
      END
    END
  END Init;

BEGIN Init
END Host.
