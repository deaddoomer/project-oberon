MODULE Input; (*NW 5.10.86 / 15.11.90 Ceres-2; PDR 21.4.12 / NW 15.5.2013 Ceres-4*)
  IMPORT Host;

  VAR
    MW, MH: INTEGER; (*mouse limits and coords*)

  PROCEDURE Available*(): INTEGER;
    RETURN Host.Available()
  END Available;

  PROCEDURE Read*(VAR ch: CHAR);
  BEGIN Host.Read(ch)
  END Read;

  PROCEDURE Mouse*(VAR keys: SET; VAR x, y: INTEGER);
  BEGIN
    Host.Mouse(keys, x, y);
    IF x < 0 THEN x := 0 ELSIF x >= MW THEN x := MW - 1 END;
    IF y < 0 THEN x := 0 ELSIF y >= MH THEN y := MH - 1 END
  END Mouse;

  PROCEDURE SetMouseLimits*(w, h: INTEGER);
  BEGIN MW := w; MH := h
  END SetMouseLimits;

  PROCEDURE Init*;
  END Init;

BEGIN Init
END Input.
