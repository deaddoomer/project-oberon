MODULE Host;  (*linux/386*)
  IMPORT SYSTEM, Modules;

  CONST
    stderr = 2;
    msMax = 2; kbdMax = 32;

  VAR
    syscall1: PROCEDURE (eax: INTEGER): INTEGER;
    syscall2: PROCEDURE (eax, ebx: INTEGER): INTEGER;
    syscall3: PROCEDURE (eax, ebx, ecx: INTEGER): INTEGER;
    syscall4: PROCEDURE (eax, ebx, ecx, edx: INTEGER): INTEGER;

    tty: INTEGER; (*/dev/tty fidle desc*)
    init: SET; (*0=vtmode, 1=kdmode, 2=kbmode*)
    vtmode, kdmode, kbmode: INTEGER; (*backup*)
    vtmodestr: ARRAY 8 OF BYTE; (*struct vt_mode*)

    kbdCode: BYTE; (*last keyboard code read*)
    Recd, Up, Shift, Ctrl, Ext: BOOLEAN;
    KTabAdr: INTEGER;  (*keyboard code translation table*)
    MX, MY: INTEGER; (*mouse coords*)
    MK: SET; (*mouse keys*)
    kbd, kbdkw: ARRAY kbdMax OF INTEGER; (*keyboard event fd*)
    ms: ARRAY msMax OF INTEGER; (*mouse fd*)
    KN, MN: INTEGER; (*number of keyboard/mouse*)

    Base, Width, Height: INTEGER;
    fb0, stride, bpp, xoff, yoff: INTEGER;

    thread: INTEGER;
    quit: BOOLEAN;

  (* --------- utility --------- *)

  PROCEDURE UnixToClock(t: INTEGER): INTEGER;
    VAR z, era, y, doe, yoe, doy, mp, second, minute, hour, day, month, year: INTEGER;
  BEGIN
    z := (t DIV 86400) + 719468;
    IF z >= 0 THEN era := z DIV 146097 ELSE era := (z - 146096) DIV 146097 END;
    doe := z - era*146097;
    yoe := (doe - doe DIV 1460 + doe DIV 36524 - doe DIV 146096) DIV 365;
    doy := doe - (365*yoe + yoe DIV 4 - yoe DIV 100);
    mp := (5*doy + 2) DIV 153;
    y := yoe + era*400;
    day := doy - (153*mp + 2) DIV 5 + 1;
    IF mp < 10 THEN month := mp + 3 ELSE month := mp - 9 END;
    IF month <= 2 THEN year := y + 1 ELSE year := y END;
    year := year MOD 100;
    hour := t DIV 3600 MOD 24;
    minute := t DIV 60 MOD 60;
    second := t MOD 60;
    RETURN ((((year*10H + month)*20H + day)*20H + hour)*40H + minute)*40H + second
  END UnixToClock;

  PROCEDURE FileInfo*(name: ARRAY OF CHAR; VAR length, date: INTEGER);
    VAR res: INTEGER; statbuf: ARRAY 24 OF INTEGER; 
  BEGIN length := -1; date := 0;
    res := syscall3(195, SYSTEM.ADR(name), SYSTEM.ADR(statbuf)); (*stat64*)
    IF res = 0 THEN
      IF (statbuf[12] = 0) & (statbuf[11] >= 0) THEN length := statbuf[11] END; (*st_size*)
      date := UnixToClock(statbuf[20]) (*st_ctime*)
    END
  END FileInfo;

  PROCEDURE Write*(fd, buf, size: INTEGER): INTEGER;
    VAR len: INTEGER;
  BEGIN
    REPEAT len := syscall4(4, fd, buf, size) UNTIL len # -4;
    RETURN len
  END Write;

  PROCEDURE WriteStringAdr*(fd, p: INTEGER);
    VAR len: INTEGER; ch: CHAR;
  BEGIN
    IF p # 0 THEN
      len := 0; SYSTEM.GET(p, ch); WHILE ch # 0X DO INC(len); SYSTEM.GET(p+len, ch) END;
      len := Write(fd, p, len)
    END
  END WriteStringAdr;

  PROCEDURE WriteString*(fd: INTEGER; str: ARRAY OF CHAR);
  BEGIN WriteStringAdr(fd, SYSTEM.ADR(str))
  END WriteString;

  PROCEDURE WriteLn*(fd: INTEGER);
    VAR s: ARRAY 2 OF CHAR;
  BEGIN s[0] := 0AX; s[1] := 0X; WriteString(fd, s)
  END WriteLn;

  (* --------- signals --------- *)

  PROCEDURE Cleanup;
    VAR res: INTEGER;
  BEGIN
    quit := TRUE;
    IF thread > 0 THEN
      REPEAT res := syscall4(7, thread, 0, 0) UNTIL res # -4;
    END;
    IF tty >= 0 THEN
      IF 2 IN init THEN
        REPEAT UNTIL syscall4(54, tty, 4B45H, kbmode) # -4; (*ioctl(tty, KDSKBMODE, kbmode)*)
      END;
      IF 1 IN init THEN
        REPEAT UNTIL syscall4(54, tty, 4B3AH, kdmode) # -4; (*ioctl(tty, KDSETMODE, kdmode)*)
      END;
      IF 0 IN init THEN
        vtmodestr[0] := vtmode;
        REPEAT UNTIL syscall4(54, tty, 5602H, SYSTEM.ADR(vtmodestr)) # -4; (*ioctl(tty, VT_SETMODE, vtmodestr)*)
      END;
      REPEAT UNTIL syscall2(6, tty) # -4; (*close(tty)*)
      tty := -1;
    END;
  END Cleanup;

  PROCEDURE Lock;
    VAR m: INTEGER;
  BEGIN
    REPEAT m := syscall1(68) UNTIL m # -4; (*siggetmask()*)
    REPEAT UNTIL syscall2(69, ORD(SYSTEM.VAL(SET, m) - {2})) # -4; (*sigsetmask(m-2)*)
  END Lock;

  PROCEDURE Unlock;
    VAR m: INTEGER;
  BEGIN
    REPEAT m := syscall1(68) UNTIL m # -4; (*siggetmask()*)
    REPEAT UNTIL syscall2(69, ORD(SYSTEM.VAL(SET, m) + {2})) # -4; (*sigsetmask(m+2)*)
  END Unlock;

  PROCEDURE Abort*;
    VAR pid, res: INTEGER;
  BEGIN
    Lock;
    Cleanup;
    REPEAT pid := syscall1(20) UNTIL pid # -4; (*getpid()*)
    REPEAT UNTIL syscall3(37, -pid, 6) # -4; (*kill(getpid(), SIGABRT)*)
    REPEAT UNTIL syscall2(1, 1) # -4; (*exit(1)*)
    REPEAT UNTIL FALSE
  END Abort;

  PROCEDURE Signal; (*void Signal(int sig, struct sigcontext ctx);*)
    VAR AbortCommand: PROCEDURE;
  BEGIN
    SYSTEM.GET(Modules.MTOrg - 32, AbortCommand);
    IF AbortCommand # NIL THEN
      SYSTEM.PUT(SYSTEM.REG(5) + 12 + 56, AbortCommand); (*ctx.eip := AbortCommand*)
    ELSE Abort
    END;
  END Signal;

  PROCEDURE InstallSignals;
    VAR sa: RECORD
      handler: INTEGER;
      mask, flags: SET;
      restorer: INTEGER;
    END;
  BEGIN
    sa.handler := SYSTEM.ADR(Signal);
    sa.mask := {};
    sa.flags := {28, 30}; (*SA_RESTART|SA_NODEFER*)
    sa.restorer := 0;
    REPEAT UNTIL syscall4(67, 2, SYSTEM.ADR(sa), 0) # -4; (*sigaction(SIGINT, &sa, NULL)*)
  END InstallSignals;

  (* --------- commands --------- *)

  PROCEDURE Sleep*(ms: INTEGER);
    VAR tv: ARRAY 2 OF INTEGER;
  BEGIN tv[0] := ms DIV 1000; tv[1] := ms MOD 1000 * 1000;
    REPEAT UNTIL syscall3(162, SYSTEM.ADR(tv), SYSTEM.ADR(tv)) # -4; (*nanosleep(&tv, &tv)*)
  END Sleep;

  PROCEDURE Idle*;
  BEGIN Sleep(1)
  END Idle;

  PROCEDURE Quit*(code: INTEGER);
  BEGIN
    REPEAT UNTIL syscall1(36) # -4; (*sync*)
    Cleanup;
    REPEAT UNTIL syscall2(1, code) # -4; (*exit(code)*)
    REPEAT UNTIL FALSE;
  END Quit;

  PROCEDURE Exit*;
  BEGIN Quit(0)
  END Exit;

  PROCEDURE Sync*;
  BEGIN REPEAT UNTIL syscall1(36) # -4 (*sync*)
  END Sync;

  (* --------- input --------- *)

  PROCEDURE Peek();
    VAR i, fd, res, count, type, code, value: INTEGER; buf: ARRAY 4 OF INTEGER;
  BEGIN i := 0;
    WHILE ~Recd & (i < KN) DO fd := kbd[i];
      REPEAT
        count := 0;
        REPEAT (*read*)
          res := syscall4(3, fd, SYSTEM.ADR(buf) + count, 16 - count);
          IF res > 0 THEN INC(count, res) END
        UNTIL (res # -4) & ((count = 0) OR (count = 16));
        IF count = 16 THEN
          type := buf[2] MOD 65536;
          code := buf[2] DIV 65536 MOD 65536;
          value := buf[3];
          IF type = 0 THEN (*EV_SYN*)
            IF code = 3 THEN (* SYN_DROPPED => fsync(fd) *)
              REPEAT UNTIL syscall2(118, fd) # -4;
              kbdCode := 0; Up := FALSE; Shift := FALSE; Ctrl := FALSE; Recd := FALSE
            END
          ELSIF (type = 1) & (code <= 127) THEN (*EV_KEY*)
            kbdCode := code; Up := value = 0;
            IF (code = 42) OR (code = 54) THEN (*shift*) Shift := ~Up
            ELSIF code = 29 THEN (*ctrl*) Ctrl := ~Up
            ELSIF ~Up THEN Recd := TRUE (*real key going down*)
            END
          END
        END
      UNTIL (res <= 0) OR (count = 0) OR Recd;
      INC(i)
    END
  END Peek;

  PROCEDURE Available*(): INTEGER;
  BEGIN Peek();
    RETURN ORD(Recd)
  END Available;

  PROCEDURE Read*(VAR ch: CHAR);
  BEGIN
    WHILE ~Recd DO Peek() END ;
    IF Shift OR Ctrl THEN INC(kbdCode, 80H) END; (*ctrl implies shift*)
  (* ch := kbdTab[kbdCode]; *)
    SYSTEM.GET(KTabAdr + kbdCode, ch);
    IF Ctrl THEN ch := CHR(ORD(ch) MOD 20H) END;
    Recd := FALSE
  END Read;

  PROCEDURE PeekMouse();
    VAR i, dx, dy, res, count, fd: INTEGER; s, u, v: SET; buf: ARRAY 3 OF BYTE;
  BEGIN u := {}; v := {};
    FOR i := 0 TO MN - 1 DO fd := ms[i];
      count := 0;
      REPEAT
        res := syscall4(3, fd, SYSTEM.ADR(buf) + count, 3 - count);
        IF res > 0 THEN INC(count, res) END
      UNTIL (res # -4) & ((count = 0) OR (count = 3));
      IF count = 3 THEN
        s := SYSTEM.VAL(SET, ORD(buf[0]));
        IF 0 IN s THEN INCL(u, 2) ELSE INCL(v, 2) END; (* ML *)
        IF 1 IN s THEN INCL(u, 0) ELSE INCL(v, 0) END; (* MR *)
        IF 2 IN s THEN INCL(u, 1) ELSE INCL(v, 1) END; (* MM *)
        dx := buf[1]; IF dx >= 128 THEN DEC(dx, 256) END;
        dy := buf[2]; IF dy >= 128 THEN DEC(dy, 256) END;
        INC(MX, dx); INC(MY, dy)
      END
    END;
    IF MX < 0 THEN MX := 0 ELSIF MX >= Width THEN MX := Width-1 END;
    IF MY < 0 THEN MY := 0 ELSIF MY >= Height THEN MY := Height-1 END;
    MK := MK - v + u
  END PeekMouse;

  PROCEDURE Mouse*(VAR keys: SET; VAR x, y: INTEGER);
    CONST modifier = 56; (*lalt*)
    VAR i, j: INTEGER; state: ARRAY 2 OF SET; Alt: BOOLEAN;
  BEGIN PeekMouse(); keys := MK; x := MX; y := MY;
    IF 2 IN MK THEN (*left*)
      Alt := FALSE; j := 0;
      FOR i := 0 TO LEN(state) - 1 DO state[i] := {} END;
      WHILE ~Alt & (j < KN) DO
        (*ioctl(fd, EVIOCGKEY(sizeof(state)), state)*)
        REPEAT i := syscall4(54, kbd[j], 80004518H + LEN(state)*4 * 65536, SYSTEM.ADR(state)) UNTIL i # -4;
        Alt := (modifier MOD 32) IN state[modifier DIV 32];
        INC(j);
      END;
      IF Alt THEN keys := MK - {2} + {1} END;
    END;
  END Mouse;

  PROCEDURE Decimal (VAR s: ARRAY OF CHAR; j, x: INTEGER);
    VAR i: INTEGER; x0: LONGINT; a: ARRAY 12 OF CHAR;
  BEGIN
    IF ROR(x, 31) = 1 THEN a := "8463847412-"; i := 12
    ELSE i := 0;
      IF x < 0 THEN x0 := -x ELSE x0 := x END;
      REPEAT
        a[i] := CHR(x0 MOD 10 + 30H); x0 := x0 DIV 10; INC(i)
      UNTIL x0 = 0;
      IF x < 0 THEN a[i] := "-"; INC(i) END
    END;
    REPEAT DEC(i); s[j] := a[i]; INC(j) UNTIL i = 0; s[j] := 0X
  END Decimal;

  PROCEDURE InitInput;
    VAR i, j, k, fd, res, key: INTEGER;
      name: ARRAY 32 OF CHAR;
      events: SET;
      ok: BOOLEAN;
      keys: ARRAY 8 OF SET;
      ch: CHAR;
  BEGIN Up := FALSE; Shift := FALSE; Ctrl := FALSE; Recd := FALSE;
    KTabAdr := SYSTEM.ADR($
      00 1B 31 32 33 34 35 36  37 38 39 30 2D 3D 08 09
      71 77 65 72 74 79 75 69  6F 70 5B 5D 0D 00 61 73
      64 66 67 68 6A 6B 6C 3B  27 60 00 5C 7A 78 63 76
      62 6E 6D 2C 2E 2F 00 00  00 20 00 1A 00 00 00 00
      00 00 00 00 00 00 00 00  00 00 00 00 00 00 00 00
      00 00 00 7F 00 00 00 00  00 00 00 00 00 00 00 00
      00 00 00 00 00 00 00 00  00 00 00 00 00 00 00 00
      00 00 00 00 00 00 00 00  00 00 00 00 00 00 00 00
      00 1B 21 40 23 24 25 5E  26 2A 28 29 5F 2B 08 09
      51 57 45 52 54 59 55 49  4F 50 7B 7D 0D 00 41 53
      44 46 47 48 4A 4B 4C 3A  22 7E 00 7C 5A 58 43 56
      42 4E 4D 3C 3E 3F 00 00  00 20 00 00 00 00 00 00
      00 00 00 00 00 00 00 00  00 00 00 00 00 00 00 00
      00 00 00 7F 00 00 00 00  00 00 00 00 00 00 00 00
      00 00 00 00 00 00 00 00  00 00 00 00 00 00 00 00
      00 00 00 00 00 00 00 00  00 00 00 00 00 00 00 00$);

    (*open keyboard events*)
    name := "/dev/input/eventX"; KN := 0; MN := 0; i := 0;
    WHILE (i < 256) & (KN < kbdMax) DO
      Decimal(name, 16, i);
      REPEAT fd := syscall4(5, SYSTEM.ADR(name), 800H, 0) UNTIL fd # -4; (*open(name, O_NONBLOCK|O_RDONLY)*)
      IF fd >= 0 THEN
        ok := FALSE;
        REPEAT res := syscall4(54, fd, 80044520H, SYSTEM.ADR(events)) UNTIL res # -4; (*ioctl(fd, EVIOCGBIT(0, 4), &events)*)
        IF (res > 0) & (1 IN events) THEN
          REPEAT res := syscall4(54, fd, 80204521H, SYSTEM.ADR(keys)) UNTIL res # -4; (*ioctl(fd, EVIOCGBIT(EV_KEY, 32), keys)*)
          IF res > 0 THEN
            j := 0;
            REPEAT (*at least one key must be supported by keyboard*)
              key := j MOD 128;
              IF key < res * 4 THEN SYSTEM.GET(KTabAdr + j, ch); ok := (ch # 0X) & ((key MOD 32) IN keys[key DIV 32]) END;
              INC(j);
            UNTIL ok OR (j >= 256);
          END;
        END;
        IF ok THEN
          kbd[KN] := fd;
          REPEAT fd := syscall4(5, SYSTEM.ADR(name), 800H, 0) UNTIL fd # -4; (*open(name, O_NONBLOCK|O_RDONLY)*)
          kbdkw[KN] := fd; (*open copy for keyboard watcher*)
          INC(KN);
        ELSE REPEAT res := syscall2(6, fd) UNTIL res # -4; (*close*)
        END;
      END;
      INC(i);
    END;
    IF KN = 0 THEN
      WriteString(stderr, "Error: no suitable keyboard found"); WriteLn(stderr);
      Abort;
    END;

    (*open PS/2 mouse*)
    REPEAT fd := syscall4(5, SYSTEM.ADR("/dev/psaux"), 800H, 0) UNTIL fd # -4;
    IF fd >= 0 THEN ms[MN] := fd; INC(MN) END;
    REPEAT fd := syscall4(5, SYSTEM.ADR("/dev/input/mice"), 800H, 0) UNTIL fd # -4;
    IF fd >= 0 THEN ms[MN] := fd; INC(MN) END;
    IF MN = 0 THEN
      WriteString(stderr, "Error: no suitable mouse found"); WriteLn(stderr);
      Abort;
    END;
  END InitInput;

  (* --------- display --------- *)

  PROCEDURE GetBuffer*(VAR base, width, height, str, depth: INTEGER; askDepth: INTEGER);
  BEGIN
    IF askDepth = 32 THEN base := Base; width := Width; height := Height; str := stride; depth := 32;
    ELSE base := 0; width := 0; height := 0; str := 0; depth := 0;
    END;
  END GetBuffer;

  PROCEDURE SetBuffer*(base, width, height, str, depth: INTEGER);
  END SetBuffer;

  PROCEDURE Repaint*(x, y, w, h: INTEGER);
  END Repaint;

  PROCEDURE InitDisplay;
    VAR res: INTEGER; buf: ARRAY 40 OF INTEGER; msg: ARRAY 64 OF CHAR;
  BEGIN
    REPEAT res := syscall4(5, SYSTEM.ADR("/dev/fb0"), 2, 0) UNTIL res # -4; (*open("/dev/fb0", O_RDRW, 0)*)
    IF res >= 0 THEN
      fb0 := res;
      REPEAT res := syscall4(54, fb0, 4602H, SYSTEM.ADR(buf)) UNTIL res # -4; (*ioctl(fb0, FBIOGET_FSCREENINFO, &finfo)*)
      IF res = 0 THEN
        stride := buf[11]; (*struct fb_fix_screeninfo -> line_length*)
        REPEAT res := syscall4(54, fb0, 4600H, SYSTEM.ADR(buf)) UNTIL res # -4; (*ioctl(fb0, FBIOGET_VSCREENINFO, &vinfo)*)
        IF res = 0 THEN
          Width := buf[0]; Height := buf[1]; xoff := buf[4]; yoff := buf[5]; bpp := buf[6];
          IF bpp = 32 THEN
            buf[0] := 0; buf[1] := stride * Height; buf[2] := 3; buf[3] := 1; buf[4] := fb0; buf[5] := 0;
            REPEAT res := syscall2(90, SYSTEM.ADR(buf)) UNTIL res # -4; (*mmap(0, size, PROT_READ | PROT_WRITE, MAP_SHARED, fb0, 0)*)
            IF (res <= -4096) OR (res >= 0) THEN Base := res; res := 0
            ELSE msg := "failed to map framebuffer"
            END
          ELSE msg := "unsupported pixel format"
          END
        ELSE msg := "failed to get variadic screen info"
        END
      ELSE msg := "failed to get fixed screen info"
      END
    ELSE msg := "failed to open /dev/fb0"
    END;
    IF res # 0 THEN
      WriteString(stderr, "Error: "); WriteString(stderr, msg); WriteLn(stderr);
      Abort;
    END;
  END InitDisplay;

  PROCEDURE KeyboardWatcher;
    VAR i, fd, res, count, type, code, value, pid: INTEGER; buf: ARRAY 4 OF INTEGER; Up, Ctrl: BOOLEAN;
  BEGIN Ctrl := FALSE;
    REPEAT pid := syscall1(20) UNTIL pid # -4; (*getpid()*)
    REPEAT
      FOR i := 0 TO KN - 1 DO
        fd := kbdkw[i];
        REPEAT
          count := 0;
          REPEAT (*read*)
            res := syscall4(3, fd, SYSTEM.ADR(buf) + count, 16 - count);
            IF res > 0 THEN INC(count, res) END
          UNTIL (res # -4) & ((count = 0) OR (count = 16));
          IF count = 16 THEN
            type := buf[2] MOD 65536;
            code := buf[2] DIV 65536 MOD 65536;
            value := buf[3];
            IF type = 0 THEN (*EV_SYN*)
              IF code = 3 THEN (* SYN_DROPPED => fsync(fd) *)
                REPEAT UNTIL syscall2(118, fd) # -4;
                Ctrl := FALSE;
              END
            ELSIF type = 1 THEN (*EV_KEY*)
              Up := value = 0;
              IF code = 97 THEN (*rctrl*) Ctrl := ~Up
              ELSIF ~Up & ((code = 411) OR (code = 119)) THEN (*break/pause*)
                REPEAT UNTIL syscall3(37, pid, 2) # -4; (*kill(getpid(), SIGINT)*)
              END
            END
          END
        UNTIL (res <= 0) OR (count = 0);
      END;
      Sleep(10);
    UNTIL quit;
    REPEAT UNTIL syscall2(1, 0) # -4; (*exit(0)*)
    REPEAT UNTIL FALSE;
  END KeyboardWatcher;

  PROCEDURE CreateThread (p: PROCEDURE; size: INTEGER);
    VAR stack, res: INTEGER; arg: ARRAY 6 OF INTEGER;
  BEGIN
    (* mmap(0, size, PROT_READ | PROT_WRITE, MAP_PRIVATE | MAP_ANONYMOUS | MAP_STACK, -1, 0); *)
    arg[0] := 0; arg[1] := size; arg[2] := 3; arg[3] := 2H + 20H + 20000H; arg[4] := -1; arg[5] := 0;
    REPEAT stack := syscall2(90, SYSTEM.ADR(arg)) UNTIL stack # -4;
    IF (stack >= -4096) & (stack < 0) THEN
      WriteString(stderr, "Error: failed to mmap thread stack"); WriteLn(stderr);
      Abort;
    END;

    (* jump to this address after on return from syscall4() *)
    SYSTEM.PUT(stack + size - 4, SYSTEM.VAL(INTEGER, p));

    (* clone(VM|FS|FILES|SIGHAND|PARENT|THREAD|IO, stack - 4) *)
    REPEAT res := syscall4(120, 100H+200H+400H+800H+8000H+10000H+80000000H, stack + size - 4, 0) UNTIL res # -4;
    IF res < 0 THEN
      WriteString(stderr, "Error: failed to create thread for keywatcher"); WriteLn(stderr);
      Abort;
    END;
    thread := res;
  END CreateThread;

  PROCEDURE Init;
    VAR res: INTEGER;
  BEGIN
    SYSTEM.PUT(SYSTEM.ADR(syscall1), SYSTEM.ADR($8B442404 CD80 C20400 6690 90$));
    SYSTEM.PUT(SYSTEM.ADR(syscall2), SYSTEM.ADR($8B442408 8B5C2404 CD80 C20800 6690 90$));
    SYSTEM.PUT(SYSTEM.ADR(syscall3), SYSTEM.ADR($8B44240C 8B5C2408 8B4C2404 CD80 C20C00 6690 90$));
    SYSTEM.PUT(SYSTEM.ADR(syscall4), SYSTEM.ADR($8B442410 8B5C240C 8B4C2408 8B542404 CD80 C21000 6690 90$));
    InstallSignals;

    (*open current TTY*)
    REPEAT tty := syscall4(5, SYSTEM.ADR("/dev/tty"), 2, 0) UNTIL tty # -4; (*open("/dev/tty", O_RDWR)*)
    IF tty < 0 THEN
      WriteString(stderr, "Error: failed to open /dev/tty"); WriteLn(stderr);
      Abort;
    END;

    (*save TTY state*)
    REPEAT res := syscall4(54, tty, 5601H, SYSTEM.ADR(vtmodestr)) UNTIL res # -4; (*ioctl(tty, VT_GETMODE, &vtmodestr)*)
    IF res # 0 THEN
      WriteString(stderr, "Error: ioctl VT_GETMODE failed"); WriteLn(stderr);
      Abort;
    END;
    vtmode := vtmodestr[0];
    INCL(init, 0);

    REPEAT res := syscall4(54, tty, 4B3BH, SYSTEM.ADR(kdmode)) UNTIL res # -4; (*ioctl(tty, KDGETMODE, &kdmode)*)
    IF res # 0 THEN
      WriteString(stderr, "Error: ioctl KDGETMODE failed"); WriteLn(stderr);
      Abort;
    END;
    INCL(init, 1);

    REPEAT res := syscall4(54, tty, 4B44H, SYSTEM.ADR(kbmode)) UNTIL res # -4; (*ioctl(tty, KDGKBMODE, &kbmode)*)
    IF res # 0 THEN
      WriteString(stderr, "Error: ioctl KDGKBMODE failed"); WriteLn(stderr);
      Abort;
    END;
    INCL(init, 2);

    InitInput;
    InitDisplay;
    MY := Height;

    (*disable automatic TTY switch*)
    vtmodestr[0] := 1; (* vtmodestr.mode = VT_PROCESS *)
    REPEAT res := syscall4(54, tty, 5602H, SYSTEM.ADR(vtmodestr)) UNTIL res # -4; (*ioctl(tty, VT_SETMODE, &vtmodestr)*)
    IF res # 0 THEN
      WriteString(stderr, "Error: ioctl VT_SETMODE failed"); WriteLn(stderr);
      Abort;
    END;

    (*disable keyboard echo to screen*)
    REPEAT res := syscall4(54, tty, 4B3AH, 1) UNTIL res # -4; (*ioctl(tty, KDSETMODE, KD_GRAPHICS)*)
    IF res # 0 THEN
      WriteString(stderr, "Error: ioctl KDSETMODE failed"); WriteLn(stderr);
      Abort;
    END;

    (*disable key processing by virtual console*)
    REPEAT res := syscall4(54, tty, 4B45H, 4) UNTIL res # -4; (*ioctl(tty, KDSKBMODE, K_OFF)*)
    IF res # 0 THEN
      WriteString(stderr, "Error: ioctl KDSKBMODE failed"); WriteLn(stderr);
      Abort;
    END;

    CreateThread(KeyboardWatcher, 4096);
  END Init;

BEGIN Init
END Host.
